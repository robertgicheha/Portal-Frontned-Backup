    LOCAL PROCEDURE fn_ProcessNewQuote@1();
    VAR
      AutoSetup@1000 : Record 110511;
      MyFile@1001 : Record 2000000022;
      CustFunc@1002 : Codeunit 110087;
      TemplatesPath@1003 : Text;
      MyFile_2@1004 : Record 2000000022;
      ERR_FOLDER_NOT_FOUND@1005 : TextConst 'ENU=Folder Name [ %1 ] not found under the following path [ %2 ]';
      FolderName@1006 : Text;
      FileManagement@1007 : Codeunit 419;
      SourceFileName@1008 : Text;
      TargetFileName@1009 : Text;
    BEGIN

      //Get Application Form Path
      SourceFileName:=FileManagement.OpenFileDialog('Application form for - [ '+UPPERCASE("Insured Name") + ' ]','' ,'');

      //Get Path
      AutoSetup.RESET;
      AutoSetup.GET;

      //Testfields
      AutoSetup.TESTFIELD("T-Drive Path");
      AutoSetup.TESTFIELD("Group Policy Templates Path");
      AutoSetup.TESTFIELD("Ind. Policy Templates Path");

      CLEAR(TemplatesPath);

      IF "Cover Type" = "Cover Type"::Individual THEN
      BEGIN
          TemplatesPath:=AutoSetup."Ind. Policy Templates Path";
      END;

      IF "Cover Type" = "Cover Type"::Group THEN
      BEGIN
          TemplatesPath:=AutoSetup."Group Policy Templates Path";
      END;


      MyFile.RESET;
      MyFile.SETRANGE(MyFile.Path,AutoSetup."T-Drive Path" + CustFunc.fn_GetMonthName(DATE2DMY("From Date",2)));   //Get Quote Month Folder
      IF MyFile.FIND('-') THEN
      BEGIN
          //Create Folder for New Quote i.e. Client Name = Folder Name
          CustFunc.fn_CreateNewFolder(AutoSetup."T-Drive Path" + CustFunc.fn_GetMonthName(DATE2DMY("From Date",2)),"Insured Name");
          FolderName:=AutoSetup."T-Drive Path" + CustFunc.fn_GetMonthName(DATE2DMY("From Date",2)) + '\'+ "Insured Name" + '\';

         //Added --> Store Folder Path in Table
          "Folder Path":=FolderName;
          "Folder Month":=CustFunc.fn_GetMonthName(DATE2DMY("From Date",2));

          MyFile_2.RESET;
          MyFile_2.SETRANGE(MyFile_2.Path,TemplatesPath);
          MyFile_2.SETRANGE(MyFile_2."Is a file",FALSE);
          IF MyFile_2.FIND('-') THEN
          BEGIN
              REPEAT
                  IF STRLEN(MyFile_2.Name) > 2 THEN
                  BEGIN
                      MyFile.RESET;
                      MyFile.SETRANGE(MyFile.Path,FolderName);
                      IF MyFile.FIND('-') THEN
                      BEGIN
                          CustFunc.fn_CreateNewFolder(FolderName,MyFile_2.Name)  //Create Respective Templates Folder
                      END ELSE
                      BEGIN
                          ERROR('Path not found');
                      END;
                  END;
              UNTIL MyFile_2.NEXT =0;
          END;

          TargetFileName:= FolderName+"Insured Name"+' - Application Form.pdf';
          CustFunc.fn_MoveFile(SourceFileName,TargetFileName);  //Copy File

      END ELSE
      BEGIN
          ERROR(ERR_FOLDER_NOT_FOUND,CustFunc.fn_GetMonthName(DATE2DMY("From Date",2)),AutoSetup."T-Drive Path");END;
    END;

    BEGIN
    END.
  }
